WARMUP = 5
STEPS = 25
SEED = 42

def make_config(n, behavior="dash", lidar=False):
    cols = math.ceil(math.sqrt(n))
    states = [[2.0 + (i % cols) * 1.2, 2.0 + (i // cols) * 1.2, 0.0] for i in range(n)]
    goals = [[p[0] + 200.0, p[1], 0.0] for p in states]
    robots = {
        "number": n, "distribution": {"name": "manual"},
        "kinematics": {"name": "diff"}, "shape": {"name": "circle", "radius": 0.15},
        "state": states, "goal": goals, "vel_max": [1.0, 1.0],
    }
    if behavior == "group_sfm":
        robots["group_behavior"] = {"name": "sfm", "vmax": 1.0, "neighbor_threshold": 5.0}
    elif behavior != "prescribed":
        robots["behavior"] = {"name": behavior}
    cfg = {"world": {"height": 400, "width": 400, "step_time": 0.1, "sample_time": 0.1, "collision_mode": "unobstructed"}, "robot": robots}
    if lidar:
        robots["sensors"] = [{"type": "lidar2d", "number": 360, "range_max": 10, "noise": False}]
        cfg["obstacle"] = {
            "number": 100, "distribution": {"name": "manual"},
            "shape": {"name": "rectangle", "length": 0.4, "width": 0.4},
            "state": [[8 + i % 10, 2 + i // 10, 0] for i in range(100)],
        }
    return cfg

def measured(name, n=0, behavior="dash", lidar=False, existing=None):
    if existing is None:
        cfg = make_config(n, behavior, lidar)
        path = OUT / (name + ".yaml")
        path.write_text(yaml.safe_dump(cfg), encoding="utf-8")
    else:
        path = ROOT / existing
    env = irsim.make(str(path), headless=True, seed=SEED, log_level="ERROR")
    explicit = [np.array([[0.3], [0.0]]) for _ in range(n)] if behavior == "prescribed" else None
    call = (lambda: env.step(explicit, action_id=[r.id for r in env.robot_list])) if explicit else env.step
    for _ in range(WARMUP):
        call()
    times = []
    for _ in range(STEPS):
        start = time.perf_counter_ns()
        call()
        times.append((time.perf_counter_ns() - start) / 1e6)
    phase = {}
    methods = ["_assign_group_action", "_objects_step", "_objects_sensor_step", "_status_step", "_contact_step", "build_tree"]
    originals = {}
    for key in methods:
        original = getattr(env, key)
        originals[key] = original
        def wrapped(*args, _key=key, _original=original, **kwargs):
            t = time.perf_counter_ns()
            try:
                return _original(*args, **kwargs)
            finally:
                phase.setdefault(_key, []).append((time.perf_counter_ns() - t) / 1e6)
        setattr(env, key, wrapped)
    for _ in range(10):
        call()
    for key, original in originals.items():
        setattr(env, key, original)
    profile = cProfile.Profile()
    profile.enable()
    for _ in range(5):
        call()
    profile.disable()
    stats = pstats.Stats(profile)
    top = []
    for (filename, lineno, funcname), (cc, nc, tt, ct, callers) in sorted(stats.stats.items(), key=lambda kv: kv[1][3], reverse=True)[:22]:
        top.append({"file": filename.replace(str(ROOT), "."), "line": lineno, "function": funcname, "calls": nc, "self_ms": round(tt*1000, 3), "cumulative_ms": round(ct*1000, 3)})
    snapshot_ms = []
    dict_ms = []
    json_ms = []
    for _ in range(7):
        t = time.perf_counter_ns()
        snap = env.get_msg()
        t2 = time.perf_counter_ns()
        data = snap.to_dict()
        t3 = time.perf_counter_ns()
        text = json.dumps(data, ensure_ascii=False, separators=(",", ":"), allow_nan=False)
        t4 = time.perf_counter_ns()
        snapshot_ms.append((t2-t)/1e6)
        dict_ms.append((t3-t2)/1e6)
        json_ms.append((t4-t3)/1e6)
    result = {
        "name": name, "robots": len(env.robot_list), "objects": len(env.objects),
        "sensors": sum(len(o.sensors) for o in env.objects),
        "warmup_steps": WARMUP, "measured_steps": STEPS, "seed": SEED,
        "scene_sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
        "step_ms": {"median": float(np.median(times)), "p95": float(np.percentile(times,95)), "min": min(times), "max": max(times), "raw": times},
        "phase_mean_ms_instrumented_separate_run": {k:statistics.mean(v) for k,v in phase.items()},
        "snapshot_median_ms": statistics.median(snapshot_ms),
        "snapshot_to_dict_median_ms": statistics.median(dict_ms),
        "dict_to_json_median_ms": statistics.median(json_ms),
        "snapshot_json_bytes": len(text.encode()),
        "profile_5_steps_top_by_cumulative": top,
    }
    if name == "contact_push_box":
        result["final_contact_pairs"] = len(env.contacts)
    if name == "lidar_4_static_100":
        # A real immutable snapshot for a later local model comprehension probe.
        (OUT/"lidar_snapshot.json").write_text(text,encoding="utf-8")
    env.end(0)
    print(json.dumps({k:v for k,v in result.items() if k not in ["profile_5_steps_top_by_cumulative"]},ensure_ascii=False),flush=True)
    return result

def kernel_memory(n):
    rand=np.random.default_rng(SEED)
    state=np.zeros((n,8))
    state[:,0:2]=rand.uniform(0,100,(n,2))
    state[:,4]=0.15
    state[:,5]=1.0
    model=SocialForceModelBatch(states=state)
    tracemalloc.start()
    t=time.perf_counter_ns()
    model.social_force()
    elapsed=(time.perf_counter_ns()-t)/1e6
    current,peak=tracemalloc.get_traced_memory()
    tracemalloc.stop()
    return {"n":n,"social_force_ms_single_instrumented":elapsed,"tracemalloc_peak_bytes":peak}
